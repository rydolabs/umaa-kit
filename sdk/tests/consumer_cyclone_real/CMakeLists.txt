cmake_minimum_required(VERSION 3.20)
project(UMAAExternalRealCycloneConsumer LANGUAGES CXX)
find_package(UMAAPrototype CONFIG REQUIRED)
if(NOT TARGET UMAA::Cyclone)
  message(FATAL_ERROR "Installed UMAA SDK is missing the real Cyclone target")
endif()
add_executable(umaa_external_real_cyclone main.cpp)
target_compile_features(umaa_external_real_cyclone PRIVATE cxx_std_17)
target_link_libraries(umaa_external_real_cyclone PRIVATE UMAA::Cyclone)
