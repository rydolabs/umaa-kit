cmake_minimum_required(VERSION 3.20)
project(UMAAServiceModuleSmoke LANGUAGES CXX)
include(CTest)
add_library(engine MODULE module.cpp)
add_library(battery MODULE module.cpp)
foreach(target engine battery)
  target_compile_features(${target} PRIVATE cxx_std_17)
  target_include_directories(${target} PRIVATE include)
endforeach()
target_compile_definitions(engine PRIVATE UMAA_SERVICE_NAME="Demo::EO::EngineControl" UMAA_SERVICE_OP=1 UMAA_RESULT_OFFSET=0)
target_compile_definitions(battery PRIVATE UMAA_SERVICE_NAME="Demo::EO::BatteryStatus" UMAA_SERVICE_OP=2 UMAA_RESULT_OFFSET=80)
add_executable(module_test test.cpp)
target_compile_features(module_test PRIVATE cxx_std_17)
target_include_directories(module_test PRIVATE include)
target_link_libraries(module_test PRIVATE ${CMAKE_DL_LIBS})
add_test(NAME module_load_unload COMMAND module_test "$<TARGET_FILE:engine>" "$<TARGET_FILE:battery>")
